cmake_minimum_required(VERSION 3.20)
project(TLBBClient VERSION 0.15.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT TLBB_LUA_SOURCE)
  set(TLBB_LUA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.tools/src/lua-5.4.9")
endif()
file(TO_CMAKE_PATH "${TLBB_LUA_SOURCE}" TLBB_LUA_SOURCE)
file(GLOB lua_sources CONFIGURE_DEPENDS "${TLBB_LUA_SOURCE}/src/*.c")
list(REMOVE_ITEM lua_sources "${TLBB_LUA_SOURCE}/src/lua.c" "${TLBB_LUA_SOURCE}/src/luac.c")
if(NOT lua_sources)
  message(FATAL_ERROR "Lua source missing; see BUILD_ENVIRONMENT.md")
endif()
add_library(tlbb_lua STATIC ${lua_sources})
target_include_directories(tlbb_lua PUBLIC "${TLBB_LUA_SOURCE}/src")
add_library(tlbb_resident_vm STATIC src/resident_vm.cpp)
target_include_directories(tlbb_resident_vm PUBLIC include)
target_compile_definitions(tlbb_resident_vm PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(tlbb_resident_vm PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(tlbb_resident_vm PUBLIC tlbb_lua)
add_library(TLBBEngine SHARED src/native_bridge.cpp src/login_bridge.cpp src/gameplay_bridge.cpp src/resident_bridge.cpp src/engine_main.cpp)
set_target_properties(TLBBEngine PROPERTIES PREFIX "")
target_include_directories(TLBBEngine PRIVATE include)
target_compile_definitions(TLBBEngine PRIVATE TLBB_ENGINE_COMBINED WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(TLBBEngine PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(TLBBEngine PRIVATE -static -static-libgcc -static-libstdc++)
target_link_libraries(TLBBEngine PRIVATE tlbb_resident_vm user32)
qt_add_executable(TLBB WIN32 src/main.cpp src/core.cpp src/worker.cpp src/launcher_window.cpp src/session_panel.cpp src/session_panel_pages.cpp src/session_panel_battle.cpp src/client_store.cpp src/builtin_resources.cpp src/route_catalog.cpp src/flow_runner.cpp src/profile_config.cpp src/official_launcher.cpp src/native_controller.cpp src/login_client.cpp tests/self_tests.cpp tests/login_tests.cpp)

add_dependencies(TLBB TLBBEngine)
target_include_directories(TLBB PRIVATE include)
target_compile_definitions(TLBB PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(TLBB PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(TLBB PRIVATE tlbb_resident_vm Qt6::Core Qt6::Gui Qt6::Widgets user32 psapi shell32 crypt32)
# Packaged releases: the same app as a DLL behind a Qt-free launcher. Only tools/build-client.ps1 turns this on, so the development build compiles once.
option(TLBB_PACKAGED "Also build the launcher and app DLL for a release" OFF)
if(TLBB_PACKAGED)
  get_target_property(tlbb_app_sources TLBB SOURCES)
  list(FILTER tlbb_app_sources EXCLUDE REGEX "\\.(rc|manifest)$")
  add_library(TLBBApp SHARED ${tlbb_app_sources})
  set_target_properties(TLBBApp PROPERTIES PREFIX "")
  target_include_directories(TLBBApp PRIVATE include)
  target_compile_definitions(TLBBApp PRIVATE TLBB_APP_DLL WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(TLBBApp PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(TLBBApp PRIVATE tlbb_resident_vm Qt6::Core Qt6::Gui Qt6::Widgets user32 psapi shell32 crypt32)
  add_executable(TLBBLauncher WIN32 src/launcher_stub.cpp)
  set_target_properties(TLBBLauncher PROPERTIES OUTPUT_NAME TLBB RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/launcher")
  target_compile_definitions(TLBBLauncher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_options(TLBBLauncher PRIVATE -static -static-libgcc -static-libstdc++ -municode)
  target_link_libraries(TLBBLauncher PRIVATE shell32 user32)
endif()
