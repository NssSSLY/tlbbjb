cmake_minimum_required(VERSION 3.20)
project(TlbbEnvironmentSmoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network)
file(TO_CMAKE_PATH "${TLBB_LUA_SOURCE}" TLBB_LUA_SOURCE)
if(NOT EXISTS "${TLBB_LUA_SOURCE}/src/lua.h")
  message(FATAL_ERROR "Set TLBB_LUA_SOURCE to the pinned Lua source directory")
endif()
file(GLOB lua_sources CONFIGURE_DEPENDS "${TLBB_LUA_SOURCE}/src/*.c")
list(REMOVE_ITEM lua_sources "${TLBB_LUA_SOURCE}/src/lua.c" "${TLBB_LUA_SOURCE}/src/luac.c")
add_library(tlbb_environment_lua STATIC ${lua_sources})
target_include_directories(tlbb_environment_lua PUBLIC "${TLBB_LUA_SOURCE}/src")
add_executable(lua_cli "${TLBB_LUA_SOURCE}/src/lua.c")
target_link_libraries(lua_cli PRIVATE tlbb_environment_lua)
set_target_properties(lua_cli PROPERTIES OUTPUT_NAME lua)
add_executable(environment_smoke main.cpp panel.ui)
set_target_properties(environment_smoke PROPERTIES AUTOMOC ON AUTOUIC ON)
target_link_libraries(environment_smoke PRIVATE tlbb_environment_lua Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network user32 psapi)
enable_testing()
add_test(NAME cpp_qt_lua_win32 COMMAND environment_smoke "${CMAKE_CURRENT_BINARY_DIR}/smoke-test.json")
set_tests_properties(cpp_qt_lua_win32 PROPERTIES TIMEOUT 15)
add_test(NAME lua_interpreter COMMAND lua_cli -e "assert(_VERSION == 'Lua 5.4'); assert(6*7 == 42)")
set_tests_properties(lua_interpreter PROPERTIES TIMEOUT 5)
